The loader creates the Try-On iframe and its provided Journey Trigger. You supply the product context and a container; the loader coordinates layout and validates messages.

## Before you start

1. Complete [access and domain setup](/start/access-keys-domains). The key, widget and API must belong to the same service environment.
2. [Prepare a garment image](/start/first-product).
3. Replace all three placeholders below: `YOUR_PUBLISHABLE_KEY`, `YOUR_PRODUCT_ID`, and the garment URL (in both the product image and configuration).

## Run the product page

1. Save the complete file below as `product.html`.
2. Authorize `localhost` on a separate development key. Your production page must use its authorized origin.
3. Serve the file's directory, for example with `python3 -m http.server 8088 --bind 127.0.0.1`.
4. Visit `http://localhost:8088/product.html`; do not open the file directly.

```html
<!doctype html>
<html lang="en">
<head>
  <meta charset="utf-8">
  <meta name="viewport" content="width=device-width, initial-scale=1">
  <title>Example garment — Try-On</title>
  <style>
    body { margin: 0; padding: 32px; font: 16px/1.6 system-ui; color: #000; background: #fff; }
    main { max-width: 800px; margin: auto; }
    .garment { display: block; max-width: 100%; width: 320px; height: 360px; object-fit: contain; }
    button { min-height: 44px; padding: 10px 20px; margin: 8px 8px 8px 0; cursor: pointer; }
    [hidden] { display: none !important; }
    :focus-visible { outline: 2px solid #9a9377; outline-offset: 3px; }
  </style>
</head>
<body>
  <main>
    <h1>Example garment</h1>
    <img class="garment" src="https://images.example.com/YOUR_GARMENT.jpg" alt="Selected garment">
    <p>Try this garment on with your own photo.</p>
    <div id="product-tryon"></div>
    <p id="status" role="status" tabindex="-1">Loading Try-On…</p>
    <button id="remove-tryon" type="button" hidden>Remove Try-On</button>
  </main>
  <script>
    const placement = document.getElementById("product-tryon");
    const status = document.getElementById("status");
    const remove = document.getElementById("remove-tryon");
    const button = document.getElementById("open-tryon");
    let api, unsubscribe = () => {}, opening = null, disposed = false;
    let disposing = false, disposeTask = null;
    let settleLoader;
    const loaderDone = new Promise(resolve => { settleLoader = resolve; });
    const script = document.createElement("script");
    script.src = "https://widget.fittingme.ai/v2/fittingme-loader.js";
    const attributes = {
      "data-api-base": "https://api.fittingme.ai",
      "data-retailer-key": "YOUR_PUBLISHABLE_KEY",
      "data-product-id": "YOUR_PRODUCT_ID",
      "data-garment-image-url": "https://images.example.com/YOUR_GARMENT.jpg",
      "data-journey": "tryon",
      "data-locale": "en",
      "data-trigger-mode": "encart",
      "data-container": "#product-tryon"
    };
    for (const [name, value] of Object.entries(attributes)) script.setAttribute(name, value);

    // This is a demo status handler, not an analytics subscription.
    function onEvent(event) {
      if (event.encart !== placement || disposed) return;
      if (event.jalon === "amorçage") {
        clearTimeout(bootTimer);
        const eligible = event.issue === "éligible";
        if (button) button.hidden = !eligible || disposing;
        status.textContent = eligible ? "Try-On is ready."
          : event.issue === "inéligible" ? "Try-On is not available for this product."
          : "Try-On could not load. Check your connection and retry by reloading.";
      }
      if (event.jalon === "ouverture") status.textContent = "Try-On opened.";
      if (event.jalon === "fermeture") {
        status.textContent = "Try-On closed.";
        if (button && !disposing && button.isConnected && !button.hidden) button.focus();
      }
      if (event.jalon === "complétion") status.textContent = "A completed Try-On was observed.";
    }
    function observe(name, command) {
      return new Promise((resolve, reject) => {
        const timer = setTimeout(() => finish(new Error("No observed " + name)), 2500);
        const off = api.on(event => {
          if (event.encart === placement && event.jalon === name) finish();
        });
        function finish(error) {
          clearTimeout(timer); off();
          if (error) reject(error); else resolve();
        }
        const verdict = command();
        if (verdict !== "accepté") finish(new Error("Command refused: " + verdict));
      });
    }
    function report(error) { status.textContent = error.message; }
    function openTryOn() {
      if (!api || disposed || disposing || opening) return;
      if (api.resolve(placement)?.tiroir_ouvert) return;
      if (button) button.disabled = true;
      opening = observe("ouverture", () => api.open(placement));
      opening.catch(report).finally(() => {
        opening = null;
        if (button) button.disabled = false;
      });
    }
    async function closeTryOn() {
      if (opening) await opening;
      if (!api?.resolve(placement)?.tiroir_ouvert) return;
      await observe("fermeture", () => api.close(placement));
      // Completion is posted after closure. Keep listeners through that task.
      await new Promise(resolve => setTimeout(resolve, 0));
    }
    function disposeTryOn() {
      if (disposed) return Promise.resolve();
      if (disposeTask) return disposeTask;
      disposing = true;
      disposeTask = (async () => {
        await loaderDone; // A late script load still belongs to this owner.
        if (opening) await opening;
        await closeTryOn();
        if (api?.resolve(placement)) {
          const verdict = api.destroy(placement);
          if (verdict !== "accepté") throw new Error("Teardown refused: " + verdict);
        }
        disposed = true;
        clearTimeout(bootTimer); unsubscribe();
        script.onload = script.onerror = null;
        script.remove(); placement.remove();
        if (button) { button.removeEventListener("click", openTryOn); button.hidden = true; }
        remove.removeEventListener("click", onRemove);
        remove.hidden = true;
        status.textContent = "Try-On removed. Reload to mount again.";
        status.focus();
      })().catch(error => {
        disposing = false; disposeTask = null;
        throw error; // Keep the owned embed in place if closing failed.
      });
      return disposeTask;
    }
    function onRemove() { disposeTryOn().catch(report); }
    const bootTimer = setTimeout(() => {
      status.textContent = "Try-On has not become ready. Check the console and network.";
    }, 20000);
    script.onerror = () => { clearTimeout(bootTimer); status.textContent = "The loader could not load."; settleLoader(); };
    script.onload = () => {
      api = window.FittingMe;
      if (!api?.resolve(placement)) {
        clearTimeout(bootTimer); status.textContent = "No embed mounted. Check configuration."; settleLoader(); return;
      }
      unsubscribe = api.on(onEvent); // Boot may be replayed synchronously here.
      remove.hidden = false;
      settleLoader();
    };
    if (button) button.addEventListener("click", openTryOn);
    remove.addEventListener("click", onRemove);
    // Retailer-owned sample controller. These are not extra FittingMe methods.
    window.tryOnExample = { open: openTryOn, close: closeTryOn, dispose: disposeTryOn };
    document.body.append(script);
  </script>
</body>
</html>
```

## Check the result

1. Wait for an eligible boot outcome. Expect “Try-On is ready” and the provided button to appear inside the embed.
2. Click the provided button once.
3. Complete the consent steps.
4. Upload a suitable test photo, confirm it and generate the try-on.
5. Close the drawer with its close control or Escape. Focus returns to the provided trigger inside the iframe. The status reports a completed Try-On when completion is announced on close. An open result is not yet that public completion event.

### Check removal

1. Select **Remove Try-On** to exercise close-before-destroy. It keeps the embed mounted on a teardown failure and returns focus to the status after removal.
2. Reload to start another embed.

This cleanup is for deliberate component removal, not an asynchronous unload handler: a full browser navigation already tears down its document.

## What the loader owns

The loader checks both the widget window and its origin before accepting messages. It manages the iframe sandbox, fullscreen placement, scroll lock and drawer coordination. Keep the container outside ancestors with transforms, filters or containment that prevent fullscreen placement.

The optional status and removal code make initialization and cleanup visible. A minimal storefront mount can use a static configured script, but production component integrations still need the lifecycle rules demonstrated here.
